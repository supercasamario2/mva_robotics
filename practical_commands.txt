# Stop existing container
docker stop mva-container

# Remove existing container
docker rm mva-container

# Rebuild image
docker build -t mva-robotics .

# Create new container
docker run -it --name mva-container mva-robotics bash

#bind mount:
docker run -it \
  --name mva-container \
  -v "$(pwd):/app" \
  -w /app \
  mva-robotics bash

docker container prune
